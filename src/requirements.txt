# Python dependencies of the label inspector.
# Suggested: Python 3.10-3.12 (on 3.14 some wheels - torch/opencv - may be missing).
#
# SYSTEM dependencies (install first; e.g. Debian/Ubuntu):
#   sudo apt install libzbar0
# Fedora:
#   sudo dnf install zbar

# --- Computer vision ---
numpy
opencv-python            # OpenCV (pre-processing, segmentation, morphology)
pillow
pyzbar                   # barcode decoding (ZBar)

# --- Deep learning (CNN) ---
torch                    # PyTorch
torchvision              # MobileNetV3 + transforms

# --- Multi-agent orchestration + LLM ---
google-adk               # Agent Development Kit (ADK)
google-genai             # Gemini SDK (used in the fallback and by the ADK)
python-dotenv            # loads .env (GOOGLE_API_KEY)

# --- API / web chat ---
fastapi
uvicorn[standard]
gunicorn                 # production/LAN server (config/gunicorn.conf.py, UvicornWorker)
python-multipart         # image upload in FastAPI

# --- Dataset generation / tooling (optional) ---
segno                    # QR codes (config/generate_dataset.py)
python-barcode           # 1D symbologies (config/generate_dataset.py)
pystrich                 # DataMatrix (config/generate_dataset.py)
roboflow                 # download real barcode datasets (config/download_roboflow.py)
